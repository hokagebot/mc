1. Bisection Method

Root Finding

Question: Write a Python program to find an approximate root using the Bisection Method.
Program
from math import *

def F(x):
    return x**3 + 3*x - 5

x0 = float(input("Enter the 1st end of interval "))
x1 = float(input("Enter the 2nd end of interval "))

while F(x0)*F(x1) > 0:
    print("Enter an interval with opposite signs")
    x0 = float(input("Enter the 1st end of interval "))
    x1 = float(input("Enter the 2nd end of interval "))

x2 = (x0+x1)/2
print("The interval is", x0, "to", x1)
e = float(input("Enter the error u want "))
i = 1

while abs(F(x2)) > e:
    x2 = (x0+x1)/2
    if F(x0)*F(x2) < 0:
        x1 = x2
    else:
        x0 = x2
    i = i+1

print("The best approximate root is", round(x2,4))

5. Newton's Forward Difference Table

Interpolation

Question: Write a Python program to construct Newton's forward difference table and estimate y for a given x.
Program
from math import *

n = int(input("Enter the number of data points "))
x = zeros(n)
y = zeros((n,n))

for i in range(n):
    x[i] = float(input("Enter x value "))
    y[i,0] = float(input("Enter y value "))

for j in range(1,n):
    for i in range(n-j):
        y[i,j] = y[i+1,j-1]-y[i,j-1]

print("The forward difference table is")
for i in range(n):
    print(x[i], end="  ")
    for j in range(n-i):
        print(round(y[i,j],4), end="  ")
    print()

xp = float(input("Enter the value of x to find y "))
h = x[1]-x[0]
p = (xp-x[0])/h
s = y[0,0]
term = 1

for i in range(1,n):
    term = term*(p-i+1)/i
    s = s+term*y[0,i]

print("The value of y is", round(s,4))

8. Lagrange Interpolation

Interpolation

Question: Write a Python program to find y using Lagrange's interpolation formula.
Program
from math import *

n = int(input("Enter the number of data points "))
x = zeros(n)
y = zeros(n)

for i in range(n):
    x[i] = float(input("Enter x value "))
    y[i] = float(input("Enter y value "))

xp = float(input("Enter the value of x to find y "))
s = 0

for i in range(n):
    temp = y[i]
    for j in range(n):
        if i != j:
            temp = temp*(xp-x[j])/(x[i]-x[j])
    s = s+temp

print("The value of y is", round(s,4))

9. Simpson's 1/3 Rule

Numerical Integration

Question: Write a Python program to evaluate an integral using Simpson's 1/3 Rule.
Program
from numpy import *
from math import *

def F(x):
    return x**2 + 2*x - 10

x0 = float(input("Enter the lower limit "))
xn = float(input("Enter the upper limit "))
n = int(input("Enter the number of subintervals (even number) "))
h = (xn-x0)/n
d = zeros((n+1,2))

for i in range(n+1):
    d[i,0] = x0+i*h
    d[i,1] = F(d[i,0])

sum1 = 0
sum2 = 0

for i in range(1,n):
    if i%2 == 0:
        sum2 = sum2+d[i,1]
    else:
        sum1 = sum1+d[i,1]

area = h/3*(d[0,1]+d[n,1]+4*sum1+2*sum2)
print("The area bounded by curve is", round(area,4))
